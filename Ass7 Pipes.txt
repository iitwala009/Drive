#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/wait.h>

int main() {
    int pipe1[2], pipe2[2];
    pid_t pid;
    char filename[256];
    char buffer[1024];
    ssize_t n;

    // Create two pipes
    if (pipe(pipe1) == -1 || pipe(pipe2) == -1) {
        perror("Pipe creation failed");
        return 1;
    }

    // Create child process
    pid = fork();

    if (pid < 0) {
        perror("Fork failed");
        return 1;
    }

    if (pid > 0) {
        // Parent process
        close(pipe1[0]);  // Close read end of pipe1
        close(pipe2[1]);  // Close write end of pipe2

        printf("Enter file pathname: ");
        fgets(filename, sizeof(filename), stdin);

        // Remove newline from filename
        filename[strcspn(filename, "\n")] = '\0';

        // Send filename to child
        write(pipe1[1], filename, strlen(filename) + 1);
        close(pipe1[1]);

        printf("\nFile contents received from child:\n\n");

        // Read file contents from child
        while ((n = read(pipe2[0], buffer, sizeof(buffer))) > 0) {
            write(STDOUT_FILENO, buffer, n);
        }

        close(pipe2[0]);
        wait(NULL);
    } 
    else {
        // Child process
        close(pipe1[1]);  // Close write end of pipe1
        close(pipe2[0]);  // Close read end of pipe2

        // Receive filename from parent
        n = read(pipe1[0], filename, sizeof(filename));
        close(pipe1[0]);

        if (n <= 0) {
            perror("Failed to receive filename");
            close(pipe2[1]);
            exit(1);
        }

        // Open the requested file
        int fd = open(filename, O_RDONLY);

        if (fd == -1) {
            perror("File opening failed");
            close(pipe2[1]);
            exit(1);
        }

        // Read file and send contents to parent
        while ((n = read(fd, buffer, sizeof(buffer))) > 0) {
            if (write(pipe2[1], buffer, n) == -1) {
                perror("Pipe write failed");
                close(fd);
                close(pipe2[1]);
                exit(1);
            }
        }

        close(fd);
        close(pipe2[1]);
        exit(0);
    }

    return 0;
}









Working of the Program
- Parent Process: Accepts the file pathname from the user and sends it to the child through Pipe 1.
- Child Process: Receives the pathname, opens the file, reads its contents and sends them back through Pipe 2.
- Parent Process: Receives the file contents from Pipe 2 and displays them on the screen.
Two pipes are used because one pipe handles parent-to-child communication and the other handles child-to-parent communication.
Code-wise Explanation
1. Header Files
#include <stdio.h>
#include <stdlib.h>

#include <unistd.h>

2. Variable Declaration
int pipe1[2], pipe2[2];

char buffer[1024];
ssize_t n;


- pipe1 and pipe2 are used for communication.
- pid stores the process ID.
- filename stores the file pathname.

3. Create Two Pipes
if (pipe(pipe1) == -1 || pipe(pipe2) == -1) {
    perror("Pipe creation failed");
    return 1;
}


- Creates two unnamed pipes.
- Pipe 1 sends data from parent to child.
- Pipe 2 sends data from child to parent.
4. Create Child Process
pid = fork();


- Creates a new child process using the fork() system call.
- Both parent and child execute separately.
5. Parent Process
if (pid > 0) {
    close(pipe1[0]);
    close(pipe2[1]);


- Executes the parent process.
- Closes the unused read end of Pipe 1 and write end of Pipe 2.
6. Accept File Pathname
printf("Enter file pathname: ");
fgets(filename, sizeof(filename), stdin);

filename[strcspn(filename, "\n")] = '\0';


- Accepts the file pathname from the user.
- Removes the newline character from the input.
7. Send Filename to Child
write(pipe1[1], filename, strlen(filename) + 1);
close(pipe1[1]);


- Sends the filename to the child through Pipe 1.
- Closes the write end after sending data.
8. Receive and Display File Contents
while ((n = read(pipe2[0], buffer, sizeof(buffer))) > 0) {
    write(STDOUT_FILENO, buffer, n);
}


- Reads file contents received from the child through Pipe 2.
- Displays the contents on standard output.
9. Child Process
else {
    close(pipe1[1]);
    close(pipe2[0]);


- Executes the child process.
- Closes the unused write end of Pipe 1 and read end of Pipe 2.
10. Receive Filename
n = read(pipe1[0], filename, sizeof(filename));
close(pipe1[0]);


- Receives the filename sent by the parent through Pipe 1.
- Closes the read end after receiving it.
11. Open the File
int fd = open(filename, O_RDONLY);


- Opens the requested file in read-only mode.
- fd stores the file descriptor.
12. Read and Send File Contents
while ((n = read(fd, buffer, sizeof(buffer))) > 0) {
    write(pipe2[1], buffer, n);
}


- Reads the file contents into the buffer.
- Sends the contents to the parent through Pipe 2.
13. Close Files and Terminate
close(fd);
close(pipe2[1]);
exit(0);


- Closes the file and pipe.
- Terminates the child process successfully.
14. Parent Waits for Child
wait(NULL);
return 0;


- Parent waits for the child process to finish.
- return 0 indicates successful program execution.

Important Viva Questions
1. What is a Pipe?

A pipe is an IPC mechanism used to transfer data between processes.

2. Why are two pipes used?

One pipe sends the filename from parent to child, and the second sends file contents from child to parent.

3. What is Full Duplex Communication?

It allows communication in both directions between two processes.

4. What is fork()?

It is a system call used to create a child process.

5. What is STDOUT_FILENO?

It represents standard output, usually the terminal screen.

6. What is the difference between Pipe and FIFO?
- Pipe: Generally used for communication between related processes.
- FIFO: Named pipe that can communicate between independent processes.
